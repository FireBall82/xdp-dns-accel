cmake_minimum_required(VERSION 3.20)
project(net_accel_dns VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
option(ENABLE_HARDENING "Enable compiler and linker security hardening" ON)
option(BUILD_FUZZERS "Build Clang/libFuzzer DNS and packet targets" OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(ENABLE_XDP "Build the optional BPF/AF_XDP packet adapter" ON)
find_package(Threads REQUIRED)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
if(ENABLE_XDP)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBBPF REQUIRED IMPORTED_TARGET libbpf)
pkg_check_modules(LIBXDP REQUIRED IMPORTED_TARGET libxdp)
# Explicit CMake settings take precedence over the development-shell defaults.
if(NOT DEFINED BPF_CLANG AND DEFINED ENV{CLANG} AND EXISTS "$ENV{CLANG}")
    set(BPF_CLANG "$ENV{CLANG}" CACHE FILEPATH "Clang executable with BPF support")
endif()
find_program(BPF_CLANG NAMES clang)
if(NOT DEFINED BPF_STRIP AND DEFINED ENV{LLVM_STRIP} AND EXISTS "$ENV{LLVM_STRIP}")
    set(BPF_STRIP "$ENV{LLVM_STRIP}" CACHE FILEPATH "LLVM strip executable")
endif()
find_program(BPF_STRIP NAMES llvm-strip)
if(NOT BPF_CLANG OR NOT BPF_STRIP)
    message(FATAL_ERROR "clang with the BPF target and llvm-strip are required")
endif()
set(BPF_SRC "${CMAKE_CURRENT_SOURCE_DIR}/bpf/xdp_filter.bpf.c")
set(BPF_OBJ "${CMAKE_CURRENT_BINARY_DIR}/xdp_filter.bpf.o")
add_custom_command(OUTPUT "${BPF_OBJ}"
    COMMAND ${CMAKE_COMMAND} -E env HARDENING_DISABLE=all "${BPF_CLANG}" -g -O2 -target bpf
            -Wall -Wextra -I${CMAKE_CURRENT_SOURCE_DIR}/include ${LIBBPF_CFLAGS}
            -c "${BPF_SRC}" -o "${BPF_OBJ}"
    COMMAND "${BPF_STRIP}" -g "${BPF_OBJ}"
    DEPENDS "${BPF_SRC}" include/common_structs.h include/vmlinux.h
    VERBATIM COMMENT "Building DNS XDP program")
add_custom_target(bpf_target ALL DEPENDS "${BPF_OBJ}")
endif()

add_library(dns_core src/dns_protocol.cpp src/upstream.cpp src/resolver_service.cpp src/request_policy.cpp src/process_security.cpp src/keyed_hash.cpp src/runtime_metrics.cpp)
target_include_directories(dns_core PUBLIC include)
target_link_libraries(dns_core PUBLIC Threads::Threads rt)
target_compile_options(dns_core PRIVATE -Wall -Wextra -Wpedantic)
if(ENABLE_SANITIZERS)
    target_compile_options(dns_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(dns_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(dns_daemon src/main.cpp)
target_link_libraries(dns_daemon PRIVATE dns_core)
target_compile_options(dns_daemon PRIVATE -Wall -Wextra -Wpedantic)
if(ENABLE_XDP)
target_compile_definitions(dns_daemon PRIVATE NET_ACCEL_XDP=1)
target_link_libraries(dns_daemon PRIVATE PkgConfig::LIBBPF PkgConfig::LIBXDP)
add_dependencies(dns_daemon bpf_target)
endif()
if(ENABLE_HARDENING)
    # Apply to userspace only; the BPF backend has its own compilation path.
    target_compile_options(dns_core PUBLIC -fstack-protector-strong -Wformat -Werror=format-security)
    target_compile_definitions(dns_core PUBLIC _GLIBCXX_ASSERTIONS)
    set_target_properties(dns_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_compile_options(dns_daemon PRIVATE -fPIE)
    target_link_options(dns_daemon PRIVATE -pie -Wl,-z,relro,-z,now,-z,noexecstack)
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    if(ENABLE_HARDENING AND CMAKE_READELF)
        add_test(NAME dns_elf_hardening COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_hardening.py" "${CMAKE_READELF}" $<TARGET_FILE:dns_daemon>)
    endif()
    add_test(NAME dns_benchmark_regressions COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/benchmark_tests.py")
    set_tests_properties(dns_benchmark_regressions PROPERTIES TIMEOUT 15)
    add_test(NAME dns_mode_options COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/mode_options.py" $<TARGET_FILE:dns_daemon> "${ENABLE_XDP}")
    add_executable(dns_security_tests tests/security_tests.cpp)
    target_link_libraries(dns_security_tests PRIVATE dns_core)
    target_compile_options(dns_security_tests PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME dns_security_regressions COMMAND dns_security_tests)
    set_tests_properties(dns_security_regressions PROPERTIES TIMEOUT 30)
    add_executable(dns_tests tests/dns_tests.cpp)
    target_link_libraries(dns_tests PRIVATE dns_core)
    if(ENABLE_XDP)
        target_link_libraries(dns_tests PRIVATE PkgConfig::LIBXDP)
        target_compile_definitions(dns_tests PRIVATE NET_ACCEL_XDP=1)
    endif()
    target_compile_options(dns_tests PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME dns_regressions COMMAND dns_tests)
    add_test(NAME dns_network_regressions COMMAND dns_tests --network)
    set_tests_properties(dns_regressions dns_network_regressions PROPERTIES TIMEOUT 30)
    find_program(TEST_UNBOUND_EXECUTABLE NAMES unbound)
    add_test(NAME dns_socket_regressions COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/socket_tests.py" $<TARGET_FILE:dns_daemon> "${TEST_UNBOUND_EXECUTABLE}")
    add_test(NAME dns_admin_regressions COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/admin_tests.py")
    add_test(NAME dns_cli_regressions COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.py")
    set_tests_properties(dns_cli_regressions PROPERTIES TIMEOUT 30 ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
    add_test(NAME dns_daily_service_regressions COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/daily_service_tests.py")
    set_tests_properties(dns_daily_service_regressions PROPERTIES TIMEOUT 30 ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
    set_tests_properties(dns_socket_regressions PROPERTIES TIMEOUT 30)
    set_tests_properties(dns_admin_regressions PROPERTIES TIMEOUT 30 ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
endif()
if(BUILD_FUZZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        message(FATAL_ERROR "BUILD_FUZZERS requires Clang with libFuzzer")
    endif()
    file(COPY tests/fuzz_corpus DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
    foreach(kind dns packet)
        add_executable(fuzz_${kind} tests/fuzz_${kind}.cpp src/dns_protocol.cpp)
        target_include_directories(fuzz_${kind} PRIVATE include)
        target_compile_options(fuzz_${kind} PRIVATE -O1 -g -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer)
        target_link_options(fuzz_${kind} PRIVATE -fsanitize=fuzzer,address,undefined)
        if(BUILD_TESTING)
            add_test(NAME fuzz_${kind}_smoke COMMAND fuzz_${kind} -runs=10000 -max_len=4096 -timeout=2 -rss_limit_mb=512 "${CMAKE_CURRENT_BINARY_DIR}/fuzz_corpus/${kind}")
            set_tests_properties(fuzz_${kind}_smoke PROPERTIES TIMEOUT 30)
        endif()
    endforeach()
endif()
include(GNUInstallDirs)
find_program(UNBOUND_EXECUTABLE NAMES unbound)
if(NOT UNBOUND_EXECUTABLE)
    set(UNBOUND_EXECUTABLE /usr/sbin/unbound)
endif()
find_program(UNBOUND_CHECKCONF NAMES unbound-checkconf)
if(NOT UNBOUND_CHECKCONF)
    set(UNBOUND_CHECKCONF /usr/sbin/unbound-checkconf)
endif()
find_program(SYSTEMCTL_EXECUTABLE NAMES systemctl)
if(NOT SYSTEMCTL_EXECUTABLE)
    set(SYSTEMCTL_EXECUTABLE /usr/bin/systemctl)
endif()
set(DNS_PYTHON_PATH "${CMAKE_CURRENT_SOURCE_DIR}/admin")
configure_file(deploy/net-accel-admin.in dnsctl @ONLY)
file(CHMOD "${CMAKE_CURRENT_BINARY_DIR}/dnsctl" PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
set(DNS_PYTHON_PATH "${CMAKE_INSTALL_FULL_LIBEXECDIR}/net_accel_dns/admin")
configure_file(deploy/net-accel-admin.in dnsctl-installed @ONLY)
install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/dnsctl-installed" DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME dnsctl)
configure_file(deploy/net-accel-dns-generator.in net-accel-dns-generator @ONLY)
configure_file(deploy/net-accel-dns-daily.service.in net-accel-dns-daily.service @ONLY)
install(PROGRAMS "${CMAKE_CURRENT_BINARY_DIR}/net-accel-dns-generator" DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/net_accel_dns)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/net-accel-dns-daily.service" DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/net_accel_dns)
install(DIRECTORY admin/ DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/net_accel_dns/admin
        PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
set(SYSTEMD_UNIT_DIR "lib/systemd/system" CACHE PATH "systemd unit installation directory")
foreach(unit prepare dns tcp)
    if(unit STREQUAL "dns")
        set(unit_name net-accel-dns)
    elseif(unit STREQUAL "tcp")
        set(unit_name net-accel-dns-tcp)
    else()
        set(unit_name net-accel-dns-${unit})
    endif()
    configure_file(deploy/${unit_name}.service.in ${unit_name}.service @ONLY)
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/${unit_name}.service" DESTINATION ${SYSTEMD_UNIT_DIR})
endforeach()
install(FILES deploy/net-accel-dns.sysusers DESTINATION lib/sysusers.d RENAME net-accel-dns.conf)
install(TARGETS dns_daemon RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(ENABLE_XDP)
    install(FILES "${BPF_OBJ}" DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/net_accel_dns)
endif()
